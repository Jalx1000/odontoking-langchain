```python
"""WhatsApp Cloud API client for agent-template.

Single-tenant: credentials come from settings, no per-call overrides needed.
"""

import io
import re
from typing import Optional

import httpx

from app.config import settings
from app.logging import logger   # ver nota al final — crear app/logging.py

_GRAPH_URL = "https://graph.facebook.com/v25.0"


def _auth_headers() -> dict[str, str]:
    return {"Authorization": f"Bearer {settings.WHATSAPP_ACCESS_TOKEN}"}


def _pid() -> str:
    return settings.WHATSAPP_PHONE_NUMBER_ID


async def send_text_message(to: str, text: str) -> dict:
    """Send a plain text message via WhatsApp Cloud API."""
    payload = {
        "messaging_product": "whatsapp",
        "recipient_type": "individual",
        "to": to,
        "type": "text",
        "text": {"preview_url": False, "body": text},
    }
    async with httpx.AsyncClient(timeout=15) as client:
        resp = await client.post(
            f"{_GRAPH_URL}/{_pid()}/messages",
            json=payload,
            headers={**_auth_headers(), "Content-Type": "application/json"},
        )
        if not resp.is_success:
            logger.error("whatsapp_send_error", to=to, status=resp.status_code, body=resp.text[:500])
        resp.raise_for_status()
        logger.info("whatsapp_text_sent", to=to, length=len(text))
        return resp.json()


async def send_interactive_message(to: str, interactive: dict) -> dict:
    """Send an interactive (button or list) message via WhatsApp Cloud API."""
    payload = {
        "messaging_product": "whatsapp",
        "recipient_type": "individual",
        "to": to,
        "type": "interactive",
        "interactive": interactive,
    }
    async with httpx.AsyncClient(timeout=15) as client:
        resp = await client.post(
            f"{_GRAPH_URL}/{_pid()}/messages",
            json=payload,
            headers={**_auth_headers(), "Content-Type": "application/json"},
        )
        if not resp.is_success:
            logger.error("whatsapp_send_error", to=to, status=resp.status_code, body=resp.text[:500])
        resp.raise_for_status()
        logger.info("whatsapp_interactive_sent", to=to, type=interactive.get("type"))
        return resp.json()


def build_interactive_payload(mensaje: str, to: str) -> Optional[dict]:
    """Detect numbered options and return an interactive payload dict, or None."""
    option_pattern = re.compile(r"\d+\)\s*([^\n]+)")
    options = option_pattern.findall(mensaje)
    if len(options) < 2:
        return None
    body_text = option_pattern.split(mensaje)[0].strip() or mensaje[:1024]
    if len(options) <= 3:
        return {
            "type": "button",
            "body": {"text": body_text[:1024]},
            "action": {
                "buttons": [
                    {"type": "reply", "reply": {"id": opt[:256], "title": opt[:20]}}
                    for opt in options[:3]
                ]
            },
        }
    return {
        "type": "list",
        "header": {"type": "text", "text": "Seleccione una opcion"},
        "body": {"text": body_text[:1024]},
        "action": {
            "button": "Ver opciones",
            "sections": [
                {
                    "title": "Opciones disponibles",
                    "rows": [
                        {"id": f"opt_{i + 1}", "title": opt[:24], "description": ""}
                        for i, opt in enumerate(options[:10])
                    ],
                }
            ],
        },
    }


async def mark_as_read(to: str, message_id: str) -> None:
    """Mark an incoming message as read (double blue ticks)."""
    payload = {
        "messaging_product": "whatsapp",
        "status": "read",
        "message_id": message_id,
    }
    try:
        async with httpx.AsyncClient(timeout=5) as client:
            resp = await client.post(
                f"{_GRAPH_URL}/{_pid()}/messages",
                json=payload,
                headers={**_auth_headers(), "Content-Type": "application/json"},
            )
            if not resp.is_success:
                logger.warning("whatsapp_mark_read_failed", to=to, status=resp.status_code)
    except Exception as e:
        logger.warning("whatsapp_mark_read_error", to=to, error=str(e))


async def send_typing_indicator(to: str) -> None:
    """Show the typing indicator while the agent processes. Best-effort."""
    payload = {
        "messaging_product": "whatsapp",
        "recipient_type": "individual",
        "to": to,
        "type": "typing",
    }
    try:
        async with httpx.AsyncClient(timeout=5) as client:
            resp = await client.post(
                f"{_GRAPH_URL}/{_pid()}/messages",
                json=payload,
                headers={**_auth_headers(), "Content-Type": "application/json"},
            )
            if resp.is_success:
                logger.debug("whatsapp_typing_sent", to=to)
            else:
                logger.debug("whatsapp_typing_not_supported", to=to, status=resp.status_code)
    except Exception as e:
        logger.debug("whatsapp_typing_error", to=to, error=str(e))


async def send_response(to: str, mensaje: str) -> None:
    """Send a WhatsApp response — interactive if options detected, plain text otherwise."""
    interactive = build_interactive_payload(mensaje, to)
    try:
        if interactive:
            await send_interactive_message(to, interactive)
        else:
            await send_text_message(to, mensaje)
    except httpx.HTTPStatusError as e:
        logger.exception("whatsapp_send_failed", to=to, status=e.response.status_code, error=str(e))
        raise


async def download_media(media_id: str) -> bytes:
    """Download media bytes from WhatsApp Cloud API."""
    async with httpx.AsyncClient(timeout=30) as client:
        meta_resp = await client.get(
            f"{_GRAPH_URL}/{media_id}",
            headers=_auth_headers(),
        )
        meta_resp.raise_for_status()
        media_url = meta_resp.json().get("url", "")
        file_resp = await client.get(media_url, headers=_auth_headers())
        file_resp.raise_for_status()
        logger.info("whatsapp_media_downloaded", media_id=media_id, bytes=len(file_resp.content))
        return file_resp.content


async def transcribe_audio(audio_bytes: bytes, mime_type: str = "audio/ogg") -> str:
    """Transcribe audio bytes using OpenAI Whisper API."""
    ext_map = {
        "audio/ogg": "ogg", "audio/mpeg": "mp3", "audio/mp4": "mp4",
        "audio/webm": "webm", "audio/wav": "wav",
    }
    ext = ext_map.get(mime_type, "ogg")
    try:
        async with httpx.AsyncClient(timeout=60) as client:
            resp = await client.post(
                "https://api.openai.com/v1/audio/transcriptions",
                headers={"Authorization": f"Bearer {settings.OPENAI_API_KEY}"},
                files={"file": (f"audio.{ext}", io.BytesIO(audio_bytes), mime_type)},
                data={"model": "whisper-1", "language": "es"},
            )
            resp.raise_for_status()
            text = resp.json().get("text", "")
            logger.info("audio_transcribed", chars=len(text))
            return text
    except Exception as e:
        logger.exception("audio_transcription_failed", error=str(e))
        return ""
```

## T-3: LLM service

**Archivo:** `app/llm.py`

**Patron de referencia:** `03.agent-production/app/services/llm/service.py` + `registry.py`

**Simplificacion respecto al monolito:** El monolito tiene un registry de multiples modelos con fallback circular complejo porque sirve a todos los tenants. El template tiene un modelo default configurable + fallback fijo a `gpt-4o-mini`.

```python
"""LLM service with tenacity retry and simple fallback for agent-template."""

import logging
from typing import Any

from langchain_core.messages import BaseMessage
from langchain_openai import ChatOpenAI
from openai import APIError, APITimeoutError, RateLimitError
from pydantic import SecretStr
from tenacity import (
    before_sleep_log,
    retry,
    retry_if_exception_type,
    stop_after_attempt,
    wait_exponential,
)

from app.config import settings
from app.logging import logger

_FALLBACK_MODEL = "gpt-4o-mini"


class LLMService:
    """Single-tenant LLM service with retry and fallback.

    Configured from settings.LLM_MODEL. On failure falls back to gpt-4o-mini.
    Tools are bound once via bind_tools() — survives model switch.
    """

    def __init__(self) -> None:
        self._model_name: str = settings.LLM_MODEL
        self._bound_tools: list = []
        self._llm: Any = self._build_model(settings.LLM_MODEL)
        logger.info("llm_service_initialized", model=settings.LLM_MODEL)

    def _build_model(self, model_name: str) -> Any:
        """Instantiate a ChatOpenAI with the project defaults."""
        return ChatOpenAI(
            model=model_name,
            api_key=SecretStr(settings.OPENAI_API_KEY),
            max_tokens=2000,  # pyright: ignore[reportCallIssue]
            temperature=0.2,
        )

    def bind_tools(self, tools: list) -> "LLMService":
        """Bind tools to the LLM. Call once after instantiation."""
        self._bound_tools = tools
        self._llm = self._llm.bind_tools(tools)
        logger.debug("tools_bound", tool_count=len(tools))
        return self

    @retry(
        stop=stop_after_attempt(settings.MAX_LLM_RETRIES),
        wait=wait_exponential(multiplier=1, min=2, max=10),
        retry=retry_if_exception_type((RateLimitError, APITimeoutError, APIError)),
        before_sleep=before_sleep_log(logger, logging.WARNING),
        reraise=True,
    )
    async def _invoke_with_retry(self, messages: list) -> BaseMessage:
        """Invoke self._llm with automatic retry on transient errors."""
        response = await self._llm.ainvoke(messages)
        return response  # type: ignore[return-value]

    async def call(self, messages: list) -> BaseMessage:
        """Call the LLM with retry. Falls back to gpt-4o-mini if the primary model fails.

        Never raises — returns an error AIMessage on complete failure so the
        worker can still ack the RabbitMQ message gracefully.
        """
        try:
            return await self._invoke_with_retry(messages)
        except Exception as primary_error:
            logger.warning(
                "llm_primary_model_failed_falling_back",
                primary_model=self._model_name,
                fallback_model=_FALLBACK_MODEL,
                error=str(primary_error),
            )
            if self._model_name == _FALLBACK_MODEL:
                logger.exception("llm_fallback_also_failed", error=str(primary_error))
                raise
            # Switch to fallback — preserve tool bindings
            fallback = self._build_model(_FALLBACK_MODEL)
            if self._bound_tools:
                fallback = fallback.bind_tools(self._bound_tools)
            try:
                response = await fallback.ainvoke(messages)
                return response  # type: ignore[return-value]
            except Exception as fallback_error:
                logger.exception("llm_fallback_failed", error=str(fallback_error))
                raise


llm_service = LLMService()
```

**Lo que el agente fork de Sprint 3 puede sobreescribir:** `settings.LLM_MODEL` via env var. Nada mas.

## T-4: Memory service

**Archivo:** `app/memory.py`

**Patron de referencia:** `03.agent-production/app/services/memory.py`

**Diferencia clave:** La collection de pgvector usa `f"{settings.TENANT_SLUG}_memory"` en vez de una collection global. Esto garantiza aislamiento si dos agentes comparten la misma DB por error (no deberian, pero mejor seguro).

El `POSTGRES_DSN` tiene el formato `postgresql://user:pass@host:5432/dbname` — hay que parsearlo para pasar los campos separados a mem0 que los espera individualmente.

```python
"""Long-term memory service using mem0 + pgvector for agent-template."""

from urllib.parse import urlparse

from mem0 import AsyncMemory

from app.config import settings
from app.logging import logger


def _parse_dsn() -> dict:
    """Parse POSTGRES_DSN into components expected by mem0 pgvector config."""
    parsed = urlparse(settings.POSTGRES_DSN)
    return {
        "dbname": parsed.path.lstrip("/"),
        "user": parsed.username or "",
        "password": parsed.password or "",
        "host": parsed.hostname or "localhost",
        "port": parsed.port or 5432,
    }


class MemoryService:
    """Async memory service per tenant, backed by pgvector via mem0."""

    def __init__(self) -> None:
        self._memory: AsyncMemory | None = None
        self._collection = f"{settings.TENANT_SLUG}_memory"

    async def _get_memory(self) -> AsyncMemory:
        if self._memory is None:
            db = _parse_dsn()
            self._memory = await AsyncMemory.from_config(
                config_dict={
                    "vector_store": {
                        "provider": "pgvector",
                        "config": {
                            "collection_name": self._collection,
                            **db,
                        },
                    },
                    "llm": {
                        "provider": "openai",
                        "config": {"model": "gpt-4o-mini"},
                    },
                    "embedder": {
                        "provider": "openai",
                        "config": {"model": "text-embedding-3-small"},
                    },
                }
            )
            logger.info("memory_service_initialized", collection=self._collection)
        return self._memory

    async def initialize(self) -> None:
        """Pre-warm the mem0 instance at startup to avoid cold-init latency on first message."""
        await self._get_memory()

    async def search(self, wa_id: str, query: str) -> str:
        """Search relevant memories for a user. Returns formatted string or empty on failure.

        Only caches in-process (no Redis dependency). Results are best-effort.
        """
        try:
            memory = await self._get_memory()
            results = await memory.search(user_id=wa_id, query=query)
            result = "\n".join([f"* {r['memory']}" for r in results.get("results", [])])
            return result
        except Exception as e:
            logger.error("memory_search_failed", wa_id=wa_id, error=str(e))
            return ""

    async def add(self, wa_id: str, messages: list[dict]) -> None:
        """Add messages to long-term memory. Fire-and-forget — never raises."""
        try:
            memory = await self._get_memory()
            await memory.add(messages, user_id=wa_id, metadata={"tenant": settings.TENANT_SLUG})
            logger.info("memory_updated", wa_id=wa_id)
        except Exception as e:
            logger.exception("memory_add_failed", wa_id=wa_id, error=str(e))


memory_service = MemoryService()
```

```python
"""Usage reporter: POST consumption metrics to the platform's internal API."""

import logging

import httpx
from tenacity import (
    before_sleep_log,
    retry,
    retry_if_exception_type,
    stop_after_attempt,
    wait_exponential,
)

from app.config import settings
from app.logging import logger


@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=2, max=8),
    retry=retry_if_exception_type((httpx.TimeoutException, httpx.NetworkError)),
    before_sleep=before_sleep_log(logger, logging.WARNING),
    reraise=False,   # best-effort: never raise, just log
)
async def _post_usage(payload: dict) -> None:
    """Internal: POST usage payload with retry. Swallowed on all failures."""
    async with httpx.AsyncClient(timeout=10) as client:
        resp = await client.post(
            f"{settings.PLATFORM_URL}/api/v1/internal/usage",
            json=payload,
            headers={
                "X-Internal-Key": settings.PLATFORM_INTERNAL_KEY,
                "Content-Type": "application/json",
            },
        )
        if not resp.is_success:
            logger.warning(
                "usage_report_http_error",
                status=resp.status_code,
                body=resp.text[:200],
            )
        resp.raise_for_status()
        logger.debug("usage_reported", tenant=settings.TENANT_SLUG)


async def report_usage(
    wa_id: str,
    tokens_input: int,
    tokens_output: int,
    cost_usd: float,
) -> None:
    """Report message processing metrics to the platform.

    Best-effort: if the platform is unreachable, we log and continue.
    Never blocks the caller — call with asyncio.create_task for fire-and-forget.
    """
    payload = {
        "tenant_slug": settings.TENANT_SLUG,
        "wa_id": wa_id,
        "tokens_input": tokens_input,
        "tokens_output": tokens_output,
        "cost_usd": cost_usd,
        "msg_processed": 1,
    }
    try:
        await _post_usage(payload)
    except Exception as e:
        # reraise=False on the retry decorator handles most cases,
        # but catch any unexpected exception at this level too.
        logger.warning("usage_report_failed_silently", wa_id=wa_id, error=str(e))
```

**Nota sobre tokens y costo:** La respuesta de `ChatOpenAI.ainvoke()` incluye `response.usage_metadata` con `input_tokens` y `output_tokens`. El `BaseAgent` debe leerlos y pasarlos a `report_usage`. El costo por token de `gpt-4o-mini` es $0.00015/1K input y $0.0006/1K output (actualizar si cambia el modelo).

```python
Fork this repo and override _tools and _system_prompt_template to
build a client-specific agent. Everything else stays the same.
"""

import asyncio
import json
import os
from datetime import datetime
from typing import Annotated, Optional
from urllib.parse import quote_plus

from langchain_core.callbacks import BaseCallbackHandler
from langchain_core.messages import (

from pydantic import BaseModel

# ---------------------------------------------------------------------------

class GraphState(BaseModel):

# BaseAgent
# ---------------------------------------------------------------------------

PostgresConnPool = AsyncConnectionPool[AsyncConnection[DictRow]]

_PROMPT_FILE = os.path.join(os.path.dirname(__file__), "..", "prompts", "system.md")
with open(_PROMPT_FILE, "r") as _f:
    _PROMPT_TEMPLATE = _f.read()


class BaseAgent:
    """LangGraph agent base class.

    Override these class-level attributes in a subclass to customize:
      - _tools: list of LangChain tools for this client
      - _system_prompt_template: override the file-based prompt (optional)

    Example (in a forked repo):
        from app.agent import BaseAgent
        from app.tools import tools

        class ClientAgent(BaseAgent):
            _tools = tools
    """

    _tools: list = []  # override in subclass with client-specific tools
    _system_prompt_template: str = _PROMPT_TEMPLATE  # override or keep file-based

    def __init__(self) -> None:
        # Bind tools to LLM once at init
        if self._tools:
            llm_service.bind_tools(self._tools)
        self.tools_by_name = {t.name: t for t in self._tools}
        self._connection_pool: Optional[PostgresConnPool] = None
        self._graph: Optional[CompiledStateGraph] = None
        self._persist_tasks: set[asyncio.Task] = set()
        logger.info(
            "base_agent_initialized",
            tenant=settings.TENANT_SLUG,
            tools=list(self.tools_by_name.keys()),
        )

    # ------------------------------------------------------------------
    # Connection pool
    # ------------------------------------------------------------------

    async def _get_connection_pool(self) -> Optional[PostgresConnPool]:
        if self._connection_pool is None:
            try:
                from urllib.parse import urlparse
                parsed = urlparse(settings.POSTGRES_DSN)
                connection_url = (
                    "postgresql://"
                    f"{quote_plus(parsed.username or '')}:{quote_plus(parsed.password or '')}"
                    f"@{parsed.hostname}:{parsed.port or 5432}{parsed.path}"
                )
                self._connection_pool = AsyncConnectionPool(
                    connection_url,
                    open=False,
                    min_size=1,
                    max_size=5,
                    reconnect_timeout=30,
                    kwargs={
                        "autocommit": True,
                        "connect_timeout": 10,
                        "prepare_threshold": None,
                        "row_factory": dict_row,
                        "keepalives": 1,
                        "keepalives_idle": 30,
                        "keepalives_interval": 10,
                        "keepalives_count": 5,
                    },
                )
                await self._connection_pool.open(wait=True, timeout=30)
                logger.info("agent_connection_pool_created", tenant=settings.TENANT_SLUG)
            except Exception as e:
                logger.exception("agent_connection_pool_failed", error=str(e))
                return None
        return self._connection_pool

    # ------------------------------------------------------------------
    # System prompt
    # ------------------------------------------------------------------

    def _build_system_prompt(self, wa_id: str) -> str:
        """Render the system prompt with runtime context."""
        current_datetime = datetime.now().strftime("%A %d %B %Y %H:%M")
        return (
            self._system_prompt_template.format(
                current_datetime=current_datetime,
                tenant_slug=settings.TENANT_SLUG,
            )
            + f"\n\n# Contexto\nwa_id del usuario actual: {wa_id}"
        )

    # ------------------------------------------------------------------
    # LangGraph nodes
    # ------------------------------------------------------------------

    async def _chat(self, state: GraphState, config: RunnableConfig) -> Command:
        wa_id = config.get("metadata", {}).get("wa_id", "unknown")
        system_prompt = self._build_system_prompt(wa_id)
        langchain_messages = [SystemMessage(content=system_prompt)] + list(state.messages)
        try:
            response_message = await llm_service._llm.ainvoke(
                langchain_messages,
                config={"callbacks": config.get("callbacks", [])},
            )
            if isinstance(response_message, AIMessage) and response_message.tool_calls:
                goto = "tool_call"
            else:
                goto = END
            return Command(update={"messages": [response_message]}, goto=goto)
        except Exception as e:
            logger.exception("agent_chat_failed", wa_id=wa_id, error=str(e))
            raise

    async def _tool_call(self, state: GraphState) -> Command:
        tool_calls = state.messages[-1].tool_calls  # type: ignore[union-attr]

        async def _execute(tc: dict) -> ToolMessage:
            try:
                result = await self.tools_by_name[tc["name"]].ainvoke(tc["args"])
            except Exception as e:
                logger.warning("tool_execution_failed", tool=tc["name"], error=str(e))
                result = json.dumps({"error": str(e)})
            return ToolMessage(content=result, name=tc["name"], tool_call_id=tc["id"])

        if len(tool_calls) == 1:
            outputs = [await _execute(tool_calls[0])]
        else:
            outputs = list(await asyncio.gather(*[_execute(tc) for tc in tool_calls]))

        return Command(update={"messages": outputs}, goto="chat")

    # ------------------------------------------------------------------
    # Graph initialization
    # ------------------------------------------------------------------

    async def _create_graph(self) -> CompiledStateGraph:
        builder = StateGraph(GraphState)
        builder.add_node("chat", self._chat, destinations=("tool_call", END))
        builder.add_node(
            "tool_call",
            self._tool_call,
            destinations=("chat",),
            retry_policy=RetryPolicy(max_attempts=3),
        )
        builder.set_entry_point("chat")
        builder.set_finish_point("chat")

        pool = await self._get_connection_pool()
        if pool:
            checkpointer = AsyncPostgresSaver(pool)
            await checkpointer.setup()
        else:
            checkpointer = None

        graph = builder.compile(checkpointer=checkpointer, name=f"{settings.TENANT_SLUG}_agent")
        logger.info("agent_graph_created", tenant=settings.TENANT_SLUG, has_checkpointer=checkpointer is not None)
        return graph

    async def _get_graph(self) -> CompiledStateGraph:
        if self._graph is None:
            self._graph = await self._create_graph()
        return self._graph

    # ------------------------------------------------------------------
    # Public interface
    # ------------------------------------------------------------------

    async def get_response(self, text: str, wa_id: str) -> str:
        """Process a WhatsApp message and return the agent text response.

        Concurrent: aget_state + memory.search run in parallel before the LLM call.
        Post-response: memory.add + report_usage run as fire-and-forget tasks.
        """
        graph = await self._get_graph()

        callbacks: list[BaseCallbackHandler] = []
        if settings.LANGFUSE_TRACING_ENABLED and settings.LANGFUSE_PUBLIC_KEY:
            from langfuse.callback import CallbackHandler
            callbacks = [CallbackHandler(
                public_key=settings.LANGFUSE_PUBLIC_KEY,
                secret_key=settings.LANGFUSE_SECRET_KEY,
                host=settings.LANGFUSE_HOST,
                tags=[f"tenant:{settings.TENANT_SLUG}"],
            )]

        config: RunnableConfig = {
            "configurable": {"thread_id": wa_id},
            "callbacks": callbacks,
            "metadata": {"wa_id": wa_id, "tenant": settings.TENANT_SLUG},
        }

        try:
            # Concurrent: state check + memory search
            state_result, memory_result = await asyncio.gather(
                graph.aget_state(config),
                memory_service.search(wa_id, text),
                return_exceptions=True,
            )
            state = state_result if not isinstance(state_result, BaseException) else None
            relevant_memory = memory_result if isinstance(memory_result, str) else ""
            if isinstance(memory_result, Exception):
                logger.warning("agent_memory_search_failed", wa_id=wa_id, error=str(memory_result))

            existing_count = len(state.values.get("messages", [])) if state and state.values else 0

            # Invoke graph
            response = await graph.ainvoke(
                input={
                    "messages": [HumanMessage(content=text)],
                    "long_term_memory": relevant_memory,
                },
                config=config,
            )

            # Extract AI response
            ai_messages = [m for m in response.get("messages", []) if isinstance(m, AIMessage) and m.content]
            if not ai_messages:
                return "Disculpe, ocurrio un error. Por favor intente de nuevo."

            last_content = ai_messages[-1].content
            if isinstance(last_content, list):
                last_content = " ".join(
                    block.get("text", "") for block in last_content if isinstance(block, dict)
                )

            # Try to extract "mensaje" field from JSON response (Odontoking-style)
            try:
                parsed = json.loads(last_content)
                mensaje = parsed.get("mensaje", last_content)
            except (json.JSONDecodeError, AttributeError):
                mensaje = last_content

            # Post-response background tasks
            usage_meta = ai_messages[-1].usage_metadata or {}
            tokens_in = usage_meta.get("input_tokens", 0)
            tokens_out = usage_meta.get("output_tokens", 0)
            # gpt-4o-mini pricing (update if model changes)
            cost = (tokens_in * 0.00015 + tokens_out * 0.0006) / 1000

            new_msgs = [m for m in response.get("messages", [])[existing_count:] if isinstance(m, BaseMessage)]
            openai_msgs = [{"role": "user" if isinstance(m, HumanMessage) else "assistant", "content": str(m.content)} for m in new_msgs]

            asyncio.create_task(memory_service.add(wa_id, openai_msgs))
            asyncio.create_task(report_usage(wa_id, tokens_in, tokens_out, cost))

            return str(mensaje)

        except Exception as e:
            logger.exception("agent_get_response_failed", wa_id=wa_id, error=str(e))
            raise

    async def close(self) -> None:
        """Await pending tasks and close DB pool on shutdown."""
        if self._persist_tasks:
            await asyncio.gather(*self._persist_tasks, return_exceptions=True)
        if self._connection_pool:
            await self._connection_pool.close()
            logger.info("agent_connection_pool_closed")


# Module-level singleton — imported by worker.py
agent = BaseAgent()
```

## T-7: RabbitMQ Worker

**Archivo:** `app/worker.py`

**Patron de referencia:** `03.agent-production/app/core/broker.py` clase `RabbitMQBroker.consume()`

```python
"""AgentWorker — RabbitMQ consumer loop for agent-template."""

import asyncio
import json
from typing import Optional

import aio_pika
import aio_pika.abc

from app.agent import agent
from app.config import settings
from app.logging import logger
from app.whatsapp_client import (
    mark_as_read,
    send_response,
    send_typing_indicator,
)

_TIMEOUT = settings.LLM_TOTAL_TIMEOUT + 30  # extra buffer beyond LLM timeout


class AgentWorker:
    """Async RabbitMQ consumer that drives the agent processing loop.

    Lifecycle:
      start() is called once from FastAPI lifespan.
      It runs until the process is stopped (asyncio.CancelledError).
    """

    def __init__(self) -> None:
        self._connection: Optional[aio_pika.abc.AbstractRobustConnection] = None
        self._exchange_name = f"wa.{settings.TENANT_SLUG}"
        self._queue_name = f"wa.{settings.TENANT_SLUG}.messages"
        self._dlx_name = f"wa.{settings.TENANT_SLUG}.dlx"
        self._dlq_name = f"wa.{settings.TENANT_SLUG}.dlq"

    async def _get_connection(self) -> aio_pika.abc.AbstractRobustConnection:
        if self._connection is None or self._connection.is_closed:
            self._connection = await aio_pika.connect_robust(settings.RABBITMQ_URL)
            logger.info("rabbitmq_connected", tenant=settings.TENANT_SLUG)
        return self._connection

    async def _setup_topology(self, channel: aio_pika.abc.AbstractChannel) -> aio_pika.abc.AbstractQueue:
        """Declare exchange, DLX, main queue, and DLQ (idempotent)."""
        exchange = await channel.declare_exchange(
            self._exchange_name,
            aio_pika.ExchangeType.DIRECT,
            durable=True,
        )
        dlx = await channel.declare_exchange(
            self._dlx_name,
            aio_pika.ExchangeType.DIRECT,
            durable=True,
        )
        queue = await channel.declare_queue(
            self._queue_name,
            durable=True,
            arguments={"x-dead-letter-exchange": self._dlx_name},
        )
        await queue.bind(exchange, routing_key="messages")
        dlq_queue = await channel.declare_queue(self._dlq_name, durable=True)
        await dlq_queue.bind(dlx, routing_key="")
        logger.info("rabbitmq_topology_ready", tenant=settings.TENANT_SLUG)
        return queue

    async def _handle_message(self, message: aio_pika.abc.AbstractIncomingMessage) -> None:
        """Process one AMQP message end-to-end."""
        async with message.process(requeue=False):   # nack sends to DLQ automatically
            try:
                data = json.loads(message.body.decode())
                wa_id: str = data.get("wa_id", "")
                text: str = data.get("text", "")
                message_id: str = data.get("message_id", "")

                if not wa_id or not text:
                    logger.warning("worker_invalid_message", data=data)
                    return  # ack and skip malformed messages

                logger.info("worker_processing_message", wa_id=wa_id, tenant=settings.TENANT_SLUG)

                # Typing indicator — best-effort, don't let it block
                try:
                    await asyncio.wait_for(send_typing_indicator(wa_id), timeout=5)
                except Exception:
                    pass

                # Run agent with total timeout
                response = await asyncio.wait_for(
                    agent.get_response(text, wa_id),
                    timeout=_TIMEOUT,
                )

                # Send response + mark as read
                await send_response(wa_id, response)
                if message_id:
                    await mark_as_read(wa_id, message_id)

                logger.info("worker_message_processed", wa_id=wa_id)

            except asyncio.TimeoutError:
                logger.exception(
                    "worker_message_timeout",
                    wa_id=data.get("wa_id", "unknown") if "data" in dir() else "unknown",
                    timeout=_TIMEOUT,
                )
                raise  # triggers nack → DLQ
            except Exception as e:
                logger.exception(
                    "worker_message_failed",
                    wa_id=data.get("wa_id", "unknown") if "data" in dir() else "unknown",
                    error=str(e),
                )
                raise  # triggers nack → DLQ

    async def start(self) -> None:
        """Main consume loop. Blocks until cancelled (runs inside FastAPI lifespan)."""
        logger.info("worker_starting", tenant=settings.TENANT_SLUG, queue=self._queue_name)
        try:
            conn = await self._get_connection()
            channel = await conn.channel()
            await channel.set_qos(prefetch_count=10)
            queue = await self._setup_topology(channel)

            async with queue.iterator() as queue_iter:
                async for message in queue_iter:
                    await self._handle_message(message)

        except asyncio.CancelledError:
            logger.info("worker_stopped", tenant=settings.TENANT_SLUG)
        except Exception as e:
            logger.exception("worker_fatal_error", error=str(e))
            raise
        finally:
            if self._connection and not self._connection.is_closed:
                await self._connection.close()
```

**Nota critica — `message.process(requeue=False)`:** aio-pika tiene un context manager `process()` que automaticamente hace ACK al salir del bloque sin excepcion, y NACK con `requeue=False` si se lanza una excepcion. Con `requeue=False`, RabbitMQ rutea el mensaje al DLX (dead-letter exchange) que declaramos. Esto es mas seguro que manejar ACK/NACK manualmente.

```python
    ]

from app.logging import logger  # noqa: E402 — must import after structlog.configure

async def lifespan(app: FastAPI):

    await memory_service.initialize()
    worker = AgentWorker()
    worker_task = asyncio.create_task(worker.start())

    try:

    except asyncio.CancelledError:
        pass
    from app.agent import agent
    await agent.close()
    logger.info("agent_stopped", tenant=settings.TENANT_SLUG)
```
